\begin{exercise}\label{ex2.4}
	If $X=\operatorname{Spec}B$ is affine, check the induced map $f^\#$ is two-sided inverse of the induced map from $A\rightarrow B$ to $\operatorname{Spec}B\rightarrow\operatorname{Spec}A$, so $\alpha$ is bijective. If $X=\left\{U_i\right\}$ is an affine cover, $A\rightarrow\Gamma\!\left(X,\mathcal{O}_X\right)\rightarrow\Gamma\!\left(U_i,\mathcal{O}_{U_i}\right)$ induces a map $f_i:U_i\rightarrow\operatorname{Spec}A$. Verify $f_i|_{U_i\cap U_j}=f_j|_{U_i\cap U_j}$ and glue them together.
\end{exercise}

\begin{exercise}\label{ex2.5}
	$\operatorname{Spec}\mathbb{Z}\supseteq(p)\supseteq(0)$. For any ring $R$, there is a unique map $\mathbb{Z}\rightarrow R$ defined by $1_\mathbb{Z}\mapsto1_R$. Since $\mathcal{O}_{\operatorname{Spec}\mathbb{Z},x}=\mathbb{Z}_{\mathfrak{p}_x}$, $f^\#_x:\mathcal{O}_{\operatorname{Spec}\mathbb{Z},f(x)}\rightarrow\mathcal{O}_{X,x}$ is unique by the universal property of localization. $\mathcal{O}_{X,x}$ is a field iff its maximal ideal has a zero preimage, so $f^\#_x$ is local.
\end{exercise}

\begin{exercise}\label{ex2.6}

\end{exercise}

\begin{exercise}\label{ex2.8}
	$\left.k[\varepsilon]\middle/\!\left(\varepsilon^2\right)\right.$ has a unique prime ideal $(\varepsilon)$. Let $f_x^\#:\mathcal{O}_{X,x}\rightarrow\left.k[\varepsilon]\middle/\!\left(\varepsilon^2\right)\right.$ be a local homomorphism. The induced map $\overline{f_x^\#}:k(x)\rightarrow k$ is a $k$-homomorphism, so $k(x)\cong k$. For each $m\in\mathfrak{m}_x$, $f_x^\#(m)=a\varepsilon$ for some $a\in k$, so $f_x^\#$ defines a map $g:\mathfrak{m}_x\rightarrow k$ by $m\mapsto a$. However, $g\!\left(\mathfrak{m}_x^2\right)=0$, so $g$ induces a map $\overline{g}:\left.\mathfrak{m}_x\middle/\mathfrak{m}_x^2\right.\rightarrow k$.
\end{exercise}

\begin{exercise}\label{ex2.9}
	$Z\cap U_i$ is irreducible in the subspace $U_i$. By \textcolor{cyan}{Atiyah-Macdonald Exercise 1.20(iv)}, $Z\cap U_i$ has a unique generic point $\zeta_i$. Use the fact that a generic point is contained in every nonempty open subset, and note that $Z\cap U_i\cap U_j$ is open in the subspace $Z\cap U_i$, so $\zeta_i\in Z\cap U_i\cap U_j$. Similarly, $\zeta_j\in Z\cap U_i\cap U_j$, so $\zeta_i=\zeta_j$.
\end{exercise}

\begin{exercise}\label{ex2.16}
	\begin{enumerate}[label=(\alph*),ref=\theexercise(\alph*)]\crefalias{enumi}{exercise}
		\item\label{ex2.16.1} Let $\left\{U_i\right\}$ be an open affine cover. Then for any $x\in X_f$, there is a $U_i\ni x$ such that $U_i\cap X_f=D\!\left(\overline{f}\right)$ is open and contains $x$.
		\item\label{ex2.16.2} $a|_{U_i\cap X_f}=0$. Also $\Gamma\!\left(U_i\cap X_f,\mathcal{O}_X|_{U_i\cap X_f}\right)=B_{f|_{U_i}}$, so $\left(f^{n_i}a\right)|_{U_i}=0$. $X$ is quasi-compact, take $\max\!\left(n_i\right)$, and use the locality axiom.
		\item\label{ex2.16.3} By \hyperref[ex2.16.1]{(a)}, $b|_{U_i\cap X_f}=\left.a_i\middle/\left(f|_{U_i}\right)^{n_i}\right.$ for some $a_i\in\Gamma\!\left(U_i\cap X_f,\mathcal{O}_{U_i\cap X_f}\right)$, so $\left(f|_{U_i}\right)^{n_i}b|_{U_i\cap X_f}=a_i$. Take $n=\max\!\left(n_i\right)$, $\left(f|_{U_i}\right)^nb|_{U_i\cap X_f}=a_i^\prime$. Therefore, $a_i^\prime|_{U_i\cap U_j\cap X_f}=a_j^\prime|_{U_i\cap U_j\cap X_f}$. Since $U_i\cap U_j$ is quasi-compact, by \hyperref[ex2.16.2]{(b)}, $\left(f|_{U_i\cap U_j}\right)^{N_{ij}}\left(a_i^\prime|_{U_i\cap U_j}-a_j^\prime|_{U_i\cap U_j}\right)=0$. Take $N=\max\!\left(N_{ij}\right)$, and let $c_i=f^Na_i^\prime\in\Gamma\!\left(U_i,\mathcal{O}_X\right)$, then $c_i$ and $c_j$ are compatible on intersection. Use the gluing axiom there is a $c\in A$ such that $c|_{X_f}=f^{n+N}b$.
		\item\label{ex2.16.4} Let $\Phi:A_f\rightarrow\Gamma\!\left(X_f,\mathcal{O}_{X_f}\right)$, defined by $\frac{a}{f^n}\mapsto \frac{a|_{X_f}}{f^n}$. $\Phi$ is injective by \hyperref[ex2.16.2]{(b)}, and it is surjective by \hyperref[ex2.16.3]{(c)}.
	\end{enumerate}
\end{exercise}
